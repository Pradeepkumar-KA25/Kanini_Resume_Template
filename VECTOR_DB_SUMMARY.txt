🚀 API ENDPOINTS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  POST /api/upload
    Upload resume → Auto-stored in ChromaDB
    Returns: session_id, resume_data, preview_html

  GET /api/resumes
    List all stored resumes
    Returns: [ { id, name, email, filename, created_at }, ... ]

  GET /api/resumes/{resume_id}
    Retrieve & regenerate templates from database
    Returns: session_id, resume_data, fresh preview_html

  GET /api/resumes/search?q=<query>
    Semantic search across all resumes
    Returns: [ { id, name, score, ... }, ... ]
    Example: q="python spring boot developer"

  DELETE /api/resumes/{resume_id}
    Delete resume from database
